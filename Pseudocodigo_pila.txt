CLASE PilaDinamica:
    ATRIBUTOS:
        Arreglo elementos
        Entero capacidad
        Entero indice_tope

    // ==========================================
    // CONSTRUCTOR
    // ==========================================
    FUNCION inicializar():
        capacidad = 10
        elementos = NUEVO Arreglo de tamaño capacidad
        indice_tope = -1
    FIN FUNCION

    // ==========================================
    // ESTADO DE LA PILA
    // ==========================================
    FUNCION estaVacia():
        RETORNAR (indice_tope == -1)
    FIN FUNCION

    // ==========================================
    // INSERTAR ELEMENTO (PUSH)
    // ==========================================
    FUNCION push(nuevo_elemento):
        // 1. Verificar si hay espacio, si no, crecer el arreglo al doble
        SI (indice_tope + 1) == capacidad ENTONCES:
            NUEVA_CAPACIDAD = capacidad * 2
            NUEVO_ARREGLO = NUEVO Arreglo de tamaño NUEVA_CAPACIDAD
            
            // Copiar los datos viejos al nuevo espacio
            PARA i DESDE 0 HASTA indice_tope HACER:
                NUEVO_ARREGLO[i] = elementos[i]
            FIN PARA
            
            elementos = NUEVO_ARREGLO
            capacidad = NUEVA_CAPACIDAD
        FIN SI
        
        // 2. Insertar el elemento en el nuevo tope
        indice_tope = indice_tope + 1
        elementos[indice_tope] = nuevo_elemento
    FIN FUNCION

    // ==========================================
    // ELIMINAR ELEMENTO DEL TOPE (POP)
    // ==========================================
    FUNCION pop():
        SI estaVacia() ENTONCES:
            RETORNAR // Opcional: Lanzar error de desbordamiento (Underflow)
        SINO:
            // Simplemente retrocedemos el indice, el dato se sobreescribirá luego
            indice_tope = indice_tope - 1
        FIN SI
    FIN FUNCION

    // ==========================================
    // OBTENER ELEMENTO DEL TOPE (TOP)
    // ==========================================
    FUNCION top():
        SI estaVacia() ENTONCES:
            Lanzar Error("La pila está vacía")
        SINO:
            RETORNAR elementos[indice_tope]
        FIN SI
    FIN FUNCION
